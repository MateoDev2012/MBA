Code examples

Replace ms-mba.up.railway.app with your own deployment URL.

Vanilla

JavaScript (browser)

// Single game
const res = await fetch('https://ms-mba.up.railway.app/api/v1/games/994732206');
const { data } = await res.json();

console.log(data.name);                 // "Blox Fruits"
console.log(data.playing);              // players online right now
console.log(data.ratings.upVotes);      // likes
console.log(data.creator.name);         // owner
console.log(data.thumbnail);            // logo URL
document.querySelector('#logo').src = data.thumbnail;

Batch — up to 50 games

const ids = [994732206, 1686885941, 4924922332];
const res = await fetch('https://ms-mba.up.railway.app/api/v1/games/batch?ids=' + ids.join(','));
const { data } = await res.json();
data.forEach(g => console.log(g.name, g.playing));

Search

const res = await fetch('https://ms-mba.up.railway.app/api/v1/search?q=blox+fruits&limit=5');
const { data } = await res.json();
data.forEach(g => console.log(g.name, g.id, g.playing));

Resolve URL to universeId

const url = 'https://www.roblox.com/games/994732206/Blox-Fruits';
const res = await fetch('https://ms-mba.up.railway.app/api/v1/resolve?url=' + encodeURIComponent(url));
const { data } = await res.json();
console.log(data.universeId);  // 994732206

Usage

import { getGame, getGameQuick, batch, search, resolveUrl, trending, getGamePasses, getBadges, getDeveloper } from 'roblox-stats-sdk';

const game = await getGame(994732206);
console.log(game.name, game.playing, game.ratings.upVotes);

const quick = await getGameQuick(994732206);
console.log(quick.thumbnail, quick.ratings.upVoteRatio);

const games = await batch([994732206, 1686885941]);
const results = await search('Blox Fruits', { limit: 5 });
const id = await resolveUrl('https://www.roblox.com/games/994732206/Blox-Fruits');
const trendingGames = await trending({ limit: 10 });
const passes = await getGamePasses(994732206, { includePrices: true });
const badges = await getBadges(994732206);
const dev = await getDeveloper(994732206);

SDK unwraps { ok, data } — methods return data directly. Full docs in extras/sdk/.

Python (requests)

import requests

BASE = "https://ms-mba.up.railway.app"

# Single game
data = requests.get(f"{BASE}/api/v1/games/994732206").json()["data"]
print(data["name"], data["playing"], data["ratings"]["upVotes"])

# Batch
data = requests.get(f"{BASE}/api/v1/games/batch?ids=994732206,1686885941").json()["data"]
for g in data: print(g["name"], g["playing"])

# Search
data = requests.get(f"{BASE}/api/v1/search?q=blox+fruits&limit=5").json()["data"]
for g in data: print(g["name"], g["id"], g["playing"])

# Resolve
data = requests.get(f"{BASE}/api/v1/resolve?url=https://www.roblox.com/games/994732206/Blox-Fruits").json()["data"]
print(data["universeId"])

PHP

$base = "https://ms-mba.up.railway.app";

$json = file_get_contents("$base/api/v1/games/994732206");
$data = json_decode($json, true)["data"];
echo $data["name"] . " — " . number_format($data["playing"]) . " playing";

// Batch
$json = file_get_contents("$base/api/v1/games/batch?ids=994732206,1686885941");
$data = json_decode($json, true)["data"];
foreach ($data as $g) echo $g["name"] . " — " . number_format($g["playing"]) . "\n";

// Search
$json = file_get_contents("$base/api/v1/search?q=blox+fruits&limit=5");
$data = json_decode($json, true)["data"];
foreach ($data as $g) echo $g["name"] . " — " . $g["playing"] . "\n";

cURL

curl https://ms-mba.up.railway.app/api/v1/games/994732206

# Many at once:
curl "https://ms-mba.up.railway.app/api/v1/games/batch?ids=994732206,1686885941"

# Search
curl "https://ms-mba.up.railway.app/api/v1/search?q=blox+fruits&limit=10"

# Resolve URL
curl "https://ms-mba.up.railway.app/api/v1/resolve?url=https://www.roblox.com/games/994732206/Blox-Fruits"

# Fill template with live values
curl -X POST https://ms-mba.up.railway.app/api/v1/render \
  -H "Content-Type: application/json" \
  -d '{"universeId":994732206,"html":"{{playing}} players in {{name}} by {{creator}}"}'
